\subsection{Trajectory model}\label{trajectory-model}

\[
\begin{aligned}
w(t) &= \mu + \delta_{\text{flight}} + M\,R(\theta)\,v(t) + e(t) \\
v_i(t) &= \mathrm{OU}(\tau_{\mathrm{slow},i}, \sigma_{\mathrm{slow},i}) + \mathrm{CAR2}(f_{0,i}, \zeta_i, \sigma_{\mathrm{osc},i}) \\
e_j(t) &= \mathrm{OU}(\tau_e, \sigma_e), \quad \text{independent across rotors} \\
\delta_{\text{flight}} &= \mathbf{1}c + r, \quad c \sim N(0, s_c^2),\; r \sim N(0, \operatorname{diag}(s_r^2)) \\
\mathrm{CAR2}&: \; \ddot{x} + 2\zeta\omega_0\dot{x} + \omega_0^2 x = \xi, \quad \omega_0 = 2\pi f_0
\end{aligned}
\]

\(M\) = mixer, columns
\([\text{common}, \text{roll}, \text{pitch}, \text{yaw}]\);
\(R(\theta)\) rotates the (common, yaw) plane.
\(S(\omega) = \tfrac{2}{f_s} H (e^{i\omega}I - \Phi)^{-1} Q (e^{-i\omega}I - \Phi^\top)^{-1} H^\top\),
\(\Phi = e^{A\Delta t}\), \(Q = P - \Phi P \Phi^\top\).

\subsection{Noise model}\label{noise-model}

\[
x_m(t) = \sum_{r}\sum_{k} A_{mrk}(t)\cos\phi_{mrk}(t) + n_m(t), \qquad
\phi_{mrk} = 2\pi k\!\int_0^t\! f_r + k\theta_r(t) + \psi_{rk}(t) + \alpha_{mrk}
\]

\[
d\nu_r = -\lambda \nu_r dt + \sqrt{2\lambda}\sigma_\nu dW_r,\quad d\theta_r = \nu_r dt,\qquad
V_\theta(\tau) = 2\sigma_\nu^2\!\left[\tfrac{|\tau|}{\lambda} - \tfrac{1-e^{-\lambda|\tau|}}{\lambda^2}\right]
\]

\[
\operatorname{Var}[\Delta\psi_{rk}] = 4\pi\gamma_{rk}|\tau| \;\Rightarrow\; R_{rk}(\tau) = \exp\!\left[-\tfrac{k^2}{2}V_\theta(\tau) - 2\pi\gamma_{rk}|\tau|\right]
\]

\[
A_{mrk}(t) = \sqrt{2P_{rk}}\left(\tfrac{f_r(t)}{f_{\mathrm{ref}}}\right)^{a/2} 10^{(g_{mr}-\bar g)/20} 10^{(G_m - \bar G)/20}, \qquad P_{rk} = 10^{p_{rk}/10}
\]

\[
S_m(f,t) = 10^{h_m/10}\Big[\tfrac1R\sum_r\big(\tfrac{f_r(t)}{f_{\mathrm{ref}}}\big)^{b} + s\Big]\,
10^{\left(\mu_F + \sum_{j=1}^{14} B_j(f)\sigma_B (Lz)_j + t_F\log_2(f/f_t)\right)/10}
\]

Regimes (Michael's): standby and cruise fits composed on the carrier;
the blend weight follows \(f_r(t)\).
